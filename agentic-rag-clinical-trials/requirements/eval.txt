# crashes with a ModuleNotFoundError on 0.4.2+ regardless of whether you
# ever touch vertexai. known upstream bug, several open issues on ragas's
# repo about it as of writing - this pin is the documented workaround
# until ragas ships a fix.

langfuse>=3  # v2's callback handler lived at a different import path
